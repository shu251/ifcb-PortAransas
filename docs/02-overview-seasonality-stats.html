3.2 Hourly to weekly bins by year

Hourly environmental records are collapsed to one value per year and day-of-year bin. All parameters are averaged over the sampled hours; precipitation is the mean hourly rate scaled to inches per day (IFCB sampling is not continuous, so a sum over sampled hours would be biased).

Code 
env_hourly <- counts_seasonal_metadata_clean |>
  select(DAY, YEAR, doy, all_of(names(env_labels))) |>
  distinct() |> # counts are in long format, so collapse to one row per hour
  mutate(doy = pmin(doy, 365), # fold leap day into Dec 31
         bin = (doy - 1) %/% bin_days + 1, # 1 = Jan 1-7, etc.
         discharge_cfs = discharge_cfs / 1000)

3.3 Mean and spread across years

Code 
env_clim <- env_year_bin |>
  group_by(name, bin) |>
  summarise(n_years = n_distinct(YEAR),
            mean = mean(value),
            sd = sd(value),
            .groups = "drop") |>
  mutate(se = sd / sqrt(n_years),
         spread = if (band_type == "se") se else sd) |>
  right_join(expand_grid(name = names(env_labels), bin = 1:ceiling(365 / bin_days)),
             by = c("name", "bin")) |> # add empty rows so gaps are explicit
  mutate(missing = is.na(mean) | n_years < min_years_bin,
         across(c(mean, sd, se, spread), ~ if_else(missing, NA_real_, .x)),
         x_doy = (bin - 1) * bin_days + (bin_days + 1) / 2, # bin midpoint
         name = factor(name, levels = names(env_labels)))

3.5 Season shading

Season boundaries come from the seasons table. Cool wraps the new year, so it is split into two rectangles.

Code 
season_rects <- seasons |>
  mutate(xmin = list(start_doy), xmax = list(end_doy)) |>
  select(season, start_doy, end_doy) |>
  pmap_dfr(function(season, start_doy, end_doy) {
    if (end_doy >= start_doy) {
      tibble(season, xmin = start_doy - 0.5, xmax = end_doy + 0.5)
    } else {
      tibble(season, xmin = c(0.5, start_doy - 0.5), xmax = c(end_doy + 0.5, 365.5)) # wraps the year
    }
  }) |>
  mutate(shade = season %in% c("Spring freshening", "Autumn low-inflow")) # alternate shading

climatology <- function(dat) {
  # dat: year, doy, value for one variable; each year is scored against the other years
  expand_grid(year = unique(dat$year), doy = 1:366) |>
    mutate(stats = map2(year, doy, \(y, d) {
      in_ref <- dat$year != y & circ_dist(dat$doy, d) <= window_days
      ref <- dat$value[in_ref]
      tibble(n_ref = length(ref),
             n_ref_years = n_distinct(dat$year[in_ref]),
             clim_median = median(ref),
             clim_mad = 1.4826 * mad(ref, constant = 1))
    })) |>
    unnest(stats)
}

7 Log-transform and detrend within each block

Log10(x + 1) tames the orders-of-magnitude range. The loess residual removes slow trends that would otherwise dominate the long-period end of the spectrum.

Defines a safe detrending function. It falls back to mean-centering when a series is too short or constant for loess.

Code 
detrend_loess <- function(hrs, y, span) {
  ok <- is.finite(hrs) & is.finite(y)
  # If fewer than 20 observations, or fewer than 3 unique entires it will skip the detrended step.
  if (sum(ok) < 20 || length(unique(y[ok])) < 3) return(y - mean(y, na.rm = TRUE))
  fit <- loess(y[ok] ~ hrs[ok], span = span, degree = 1)
  out <- rep(NA_real_, length(y))
  out[ok] <- residuals(fit)
  out
}

## Hourly to weekly bins by year

Hourly environmental records are collapsed to one value per year and day-of-year bin. All parameters are averaged over the sampled hours; precipitation is the mean hourly rate scaled to inches per day (IFCB sampling is not continuous, so a sum over sampled hours would be biased).

```{r}
env_hourly <- counts_seasonal_metadata_clean |>
  select(DAY, YEAR, doy, all_of(names(env_labels))) |>
  distinct() |> # counts are in long format, so collapse to one row per hour
  mutate(doy = pmin(doy, 365), # fold leap day into Dec 31
         bin = (doy - 1) %/% bin_days + 1, # 1 = Jan 1-7, etc.
         discharge_cfs = discharge_cfs / 1000)
```

```{r}
env_year_bin <- env_hourly |>
```

## Mean and spread across years

```{r}
env_clim <- env_year_bin |>
  group_by(name, bin) |>
  summarise(n_years = n_distinct(YEAR),
            mean = mean(value),
            sd = sd(value),
            .groups = "drop") |>
  mutate(se = sd / sqrt(n_years),
         spread = if (band_type == "se") se else sd) |>
  right_join(expand_grid(name = names(env_labels), bin = 1:ceiling(365 / bin_days)),
             by = c("name", "bin")) |> # add empty rows so gaps are explicit
  mutate(missing = is.na(mean) | n_years < min_years_bin,
         across(c(mean, sd, se, spread), ~ if_else(missing, NA_real_, .x)),
         x_doy = (bin - 1) * bin_days + (bin_days + 1) / 2, # bin midpoint
         name = factor(name, levels = names(env_labels)))
env_clim |> filter(missing) |> count(name)
```

## Season shading

Season boundaries come from the `seasons` table. Cool wraps the new year, so it is split into two rectangles.

```{r}
season_rects <- seasons |>
  mutate(xmin = list(start_doy), xmax = list(end_doy)) |>
  select(season, start_doy, end_doy) |>
```

```{r}
circ_dist <- function(a, b, n = 366) pmin(abs(a - b), n - abs(a - b))
```

```{r}
min_ref_years <- 4 # minimum distinct other years in the window to score a day

climatology <- function(dat) {
  # dat: year, doy, value for one variable; each year is scored against the other years
  expand_grid(year = unique(dat$year), doy = 1:366) |>
    mutate(stats = map2(year, doy, \(y, d) {
      in_ref <- dat$year != y & circ_dist(dat$doy, d) <= window_days
      ref <- dat$value[in_ref]
      tibble(n_ref = length(ref),
             n_ref_years = n_distinct(dat$year[in_ref]),
             clim_median = median(ref),
             clim_mad = 1.4826 * mad(ref, constant = 1))
    })) |>
    unnest(stats)
}
```

Score each parameters based on a z score within each season.
```{r}
scored <- counts_seasonal_metadata_clean |>
  select(season = SEASON, DATE, doy, year = YEAR, TEMP, SAL, PRECIP_in, gage_height_ft, discharge_cfs) |> 
  distinct() |> 
  pivot_longer(cols = -c(season, DATE, year, doy)) |> 
  drop_na(value) |> 
  filter(name %in% anomaly_vars) |>
  group_by(name) |>
  group_modify(function(dat, key) {
    clim <- climatology(dat)
      dat |>
        left_join(clim, by = c("year", "doy")) |>
  mutate(robust_z = if_else(n_ref_years >= min_ref_years & clim_mad > 0,
                            (value - clim_median) / clim_mad, NA_real_))
      }) |>
  ungroup() |>
  left_join(select(thresholds, season, name, p05, p95),
            by = c("season", "name")) |>
  mutate(flag_z = abs(robust_z) > z_cut,
    flag_seasonal = value < p05 | value > p95,
    direction = case_when(
      robust_z > 0 ~ "high", robust_z < 0 ~ "low",
      TRUE ~ NA_character_))

```

* Climate median and mad represent the typical spread for the values in question. The MAD is the median absolute deviation.
* robust z is a z-score, with (value - clim_median)/climate_mad, it considers the number of days of a specific value. 
* when the value is outside the p05-p95 value, it is noted as "TRUE" in flag seasonal and the direction shows if the value is higher or lower than the expected seasonal value.

To look for seasonal anomalies, subset for `flag_seasonal == TRUE`.

# Log-transform and detrend within each block
 
Log10(x + 1) tames the orders-of-magnitude range. The loess residual removes slow trends that would otherwise dominate the long-period end of the spectrum.
 
Defines a safe detrending function. It falls back to mean-centering when a series is too short or constant for loess.
 
```{r}
#| label: detrend-function
detrend_loess <- function(hrs, y, span) {
  ok <- is.finite(hrs) & is.finite(y)
  # If fewer than 20 observations, or fewer than 3 unique entires it will skip the detrended step.
  if (sum(ok) < 20 || length(unique(y[ok])) < 3) return(y - mean(y, na.rm = TRUE))
  fit <- loess(y[ok] ~ hrs[ok], span = span, degree = 1)
  out <- rep(NA_real_, length(y))
  out[ok] <- residuals(fit)
  out
}
```
 
Detrending step. This takes individual taxa and all the count data and performs a detrending step. Output is detrended log counts and there will be one block per taxa. Use this for periodicity and recurrence trend mining. 

```{r}
#| label: detrend
detrended <- count_series |>
```
